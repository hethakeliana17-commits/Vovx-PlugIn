name: Build Vocal Tune

# Runs automatically every time you upload changes, or when you click "Run workflow" on the Actions tab.
# Push a tag like v1.0.0 to also publish the downloads on the repository's Releases page.
on:
  push:
    branches: [ main, master ]
    tags: [ 'v*' ]
  pull_request:
  workflow_dispatch:

permissions:
  contents: write

jobs:
  # Optional quick check of the sound engine. It never blocks the plugin build.
  engine-test:
    name: Engine test
    runs-on: ubuntu-latest
    continue-on-error: true
    steps:
      - uses: actions/checkout@v4
      - name: Build and run the engine test
        run: |
          g++ -O2 -std=c++17 -I . render_test.cpp -o render_test
          ./render_test

  build:
    name: Build (${{ matrix.name }})
    runs-on: ${{ matrix.os }}
    strategy:
      fail-fast: false
      matrix:
        include:
          - { name: Windows, os: windows-latest }
          - { name: macOS,   os: macos-latest }
          - { name: Linux,   os: ubuntu-latest }

    steps:
      - uses: actions/checkout@v4

      - name: Install Linux dependencies
        if: runner.os == 'Linux'
        run: |
          sudo apt-get update
          sudo apt-get install -y libasound2-dev libjack-jackd2-dev libfreetype-dev libfontconfig1-dev \
            libx11-dev libxcomposite-dev libxcursor-dev libxext-dev libxinerama-dev libxrandr-dev \
            libxrender-dev libglu1-mesa-dev mesa-common-dev

      - name: Configure
        run: cmake -B build -DCMAKE_BUILD_TYPE=Release

      - name: Build
        run: cmake --build build --config Release --parallel 4

      - name: Package (Windows)
        if: runner.os == 'Windows'
        shell: pwsh
        run: Compress-Archive -Path "build/VocalTune_artefacts/Release/*" -DestinationPath "VocalTune-${{ matrix.name }}.zip"

      - name: Package (macOS and Linux)
        if: runner.os != 'Windows'
        run: |
          cd build/VocalTune_artefacts/Release
          zip -r -y "$GITHUB_WORKSPACE/VocalTune-${{ matrix.name }}.zip" .

      - name: Upload download
        uses: actions/upload-artifact@v4
        with:
          name: VocalTune-${{ matrix.name }}
          path: VocalTune-${{ matrix.name }}.zip
          if-no-files-found: error

      - name: Publish release (only when you push a version tag)
        if: startsWith(github.ref, 'refs/tags/v')
        uses: softprops/action-gh-release@v2
        with:
          files: VocalTune-${{ matrix.name }}.zip
